cmake_minimum_required(VERSION 3.10)
project(go1_remote_control LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GO1_SIM_ONLY "Build without the real robot SDK" OFF)
if(WIN32)
  set(GO1_SIM_ONLY ON)
endif()
add_executable(go1_receiver robot/receiver.cpp)
target_compile_options(go1_receiver PRIVATE -Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
target_link_libraries(go1_receiver PRIVATE Threads::Threads)
if(BUILD_TESTING)
  add_executable(go1_policy_test tests/test_control_policy.cpp)
  target_include_directories(go1_policy_test PRIVATE robot)
  target_compile_options(go1_policy_test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME held_motion_policy COMMAND go1_policy_test)
  set_tests_properties(held_motion_policy PROPERTIES TIMEOUT 5)
endif()
if(GO1_SIM_ONLY)
  target_compile_definitions(go1_receiver PRIVATE GO1_SIM_ONLY)
else()
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Live SDK adapter requires Linux")
  endif()
  set(UNITREE_SDK_DIR "$ENV{HOME}/unitree_legged_sdk" CACHE PATH "Go1 SDK 3.8.6")
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    set(SDK_ARCH arm64)
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    set(SDK_ARCH amd64)
  else()
    message(FATAL_ERROR "Unsupported SDK architecture")
  endif()
  target_include_directories(go1_receiver PRIVATE "${UNITREE_SDK_DIR}/include")
  target_link_libraries(go1_receiver PRIVATE "${UNITREE_SDK_DIR}/lib/cpp/${SDK_ARCH}/libunitree_legged_sdk.a" Threads::Threads)
  if(BUILD_TESTING)
    add_executable(go1_sdk_lifecycle tests/test_sdk_lifecycle.cpp)
    target_compile_options(go1_sdk_lifecycle PRIVATE -Wall -Wextra -Wpedantic)
    target_include_directories(go1_sdk_lifecycle PRIVATE robot "${UNITREE_SDK_DIR}/include")
    target_link_libraries(go1_sdk_lifecycle PRIVATE "${UNITREE_SDK_DIR}/lib/cpp/${SDK_ARCH}/libunitree_legged_sdk.a" Threads::Threads)
    add_test(NAME sdk_loopback_lifecycle COMMAND go1_sdk_lifecycle)
    set_tests_properties(sdk_loopback_lifecycle PROPERTIES TIMEOUT 15)
  endif()
endif()
if(WIN32)
  target_link_libraries(go1_receiver PRIVATE ws2_32)
endif()
